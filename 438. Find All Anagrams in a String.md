# 438. Find All Anagrams in a String

```python
class Solution:
    def findAnagrams(self, s: str, p: str) -> List[int]:
        # 1. target is p, if p's length > s, return False
        if len(p) > len(s):
            return []

        # 2. store p into a [0] *26 array as the target
        target = [0] * 26
        for each_char in p:
            target[ord(each_char) - ord("a")] += 1

        # 3 check the first len(p) in s is it eaqual to target, add it to the res list
        
        window = [0] * 26
        for each_char in s[0: len(p)]:
             window[ord(each_char) - ord("a")] += 1
        
        res = []
        if window == target:
            res.append(0)
        # loop through the rest of the str s, keep the window length == len(p)
        end = len(p) 
        while end < len(s):
            window[ord(s[end]) -ord("a")] += 1
            window[ord(s[end-len(p)]) - ord("a")] -= 1
            end += 1
            if window == target:
                res.append(end-len(p))
        return res
```
