```python
class Solution:
    def checkInclusion(self, s1: str, s2: str) -> bool:
        # 1. the the length of s1 and s2
        # 2. if s1 > s2 , return false
        if len(s1) > len(s2):
            return False
        
        # 3. create a targeted list to store the counts each characters has show 
        target = [0] * 26
        for s in s1:
            target[ord(s) - ord("a")] += 1

        # 4. create a window list to save the first n character of s2
        window = [0] * 26
        for s in s2[:len(s1)]:
            window[ord(s) - ord("a")] += 1
        # 5. if window == target return True
        if window == target: 
            return True
        # 6. use end pointer to iterate from n to len(s2) -1
        end = len(s1)
        while end < len(s2):
        #   add the new character into window
            window[ord(s2[end]) - ord("a")] += 1
        #   remove the left widndows
            window[ord(s2[end-len(s1)]) -ord("a")] -= 1

        #   if window == target, return True
            if window == target:
                return True
            end += 1
        # 7. After iteration return False
        return False
```
