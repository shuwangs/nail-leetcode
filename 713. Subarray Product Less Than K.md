# 713. Subarray Product Less Than K

---

```python
class Solution:
    def numSubarrayProductLessThanK(self, nums: List[int], k: int) -> int:
        # set start and pointer:
        # set the res = 0
        # product = 1
        # start = 0, end = 0
        res  = 0
        product = 1
        start = 0
        end = 0

        # then use end to loop through the arrays
        # product = product* nums[end]
        # check if the product > k && start < end:
            # if yes, prodcut / the start vaule
            # start += 1
        #es = res + (end - start) +
        for end in range(len(nums)):
            product *= nums[end]
            while product >= k and start <= end:
                product /= nums[start]
                start += 1
            res += end - start + 1

        return res
```
